import type { RefImage } from '@shared/types'

const ALLOWED: RefImage['mime'][] = ['image/jpeg', 'image/png', 'image/webp']
const MAX_LONG_EDGE = 2048
const QUALITY = 0.92

function nanoid(): string {
  return Math.random().toString(36).slice(2, 10) + Date.now().toString(36).slice(-4)
}

export async function fileToRefImage(file: File): Promise<RefImage> {
  if (!ALLOWED.includes(file.type as RefImage['mime'])) {
    throw new Error(`不支持的图片格式：${file.type || file.name}`)
  }
  const dataUrl = await readAsDataURL(file)
  const img = await loadImage(dataUrl)
  const { canvas, w, h } = downscale(img, MAX_LONG_EDGE)
  const out = canvas.toDataURL('image/jpeg', QUALITY)
  return {
    id: nanoid(),
    index: 0, // assigned by caller after re-numbering
    name: file.name,
    mime: 'image/jpeg',
    dataUrl: out,
    width: w,
    height: h,
    bytes: approxBytes(out),
  }
}

function readAsDataURL(file: File): Promise<string> {
  return new Promise((resolve, reject) => {
    const r = new FileReader()
    r.onerror = () => reject(new Error('读取图片失败'))
    r.onload = () => resolve(r.result as string)
    r.readAsDataURL(file)
  })
}

function loadImage(src: string): Promise<HTMLImageElement> {
  return new Promise((resolve, reject) => {
    const img = new Image()
    img.onload = () => resolve(img)
    img.onerror = () => reject(new Error('图片解码失败'))
    img.src = src
  })
}

function downscale(img: HTMLImageElement, maxLong: number) {
  const long = Math.max(img.naturalWidth, img.naturalHeight)
  if (long <= maxLong) {
    const canvas = document.createElement('canvas')
    canvas.width = img.naturalWidth
    canvas.height = img.naturalHeight
    const ctx = canvas.getContext('2d')!
    ctx.drawImage(img, 0, 0)
    return { canvas, w: img.naturalWidth, h: img.naturalHeight }
  }
  const ratio = maxLong / long
  const w = Math.round(img.naturalWidth * ratio)
  const h = Math.round(img.naturalHeight * ratio)
  const canvas = document.createElement('canvas')
  canvas.width = w
  canvas.height = h
  const ctx = canvas.getContext('2d')!
  ctx.drawImage(img, 0, 0, w, h)
  return { canvas, w, h }
}

function approxBytes(dataUrl: string): number {
  const i = dataUrl.indexOf(',')
  const b64 = i >= 0 ? dataUrl.slice(i + 1) : dataUrl
  return Math.floor(b64.length * 0.75)
}

/** 把纯文本里的 `@图N` (N 是数字) 替换成高亮 HTML，
 *  把 `⟦skill:NAME⟧` 替换成红色 Skill 胶囊。其它字符 escaped。
 *  注意顺序：先 escape → skill token（含 ⟦⟧，escape 不动它们）→ @图N。 */
export function mentionizeMarkup(plain: string): string {
  if (!plain) return ''
  const escaped = plain
    .replace(/&/g, '&amp;')
    .replace(/</g, '&lt;')
    .replace(/>/g, '&gt;')
  // Skill 胶囊：只显示名字（全文生成时后台拼入）。
  // 名字已在入库时去掉 HTML 特殊字符（见 normalizeSkillName），直接匹配即可。
  const withSkills = escaped.replace(
    /⟦skill:([^\]⟧\r\n]+)⟧/g,
    (_m, n) => {
      const name = String(n).replace(/["']/g, '')
      return `<span class="skill-chip" data-skill="${name}">◆ ${name}</span>`
    },
  )
  return withSkills.replace(
    /@(?:参考图|图)(\d+)/g,
    (_m, n) => `<span class="mention" data-mention="${n}">@图${n}</span>`,
  )
}

/** 从 contenteditable 的 innerHTML 提取纯文本。
 *
 *  IMPORTANT: this is the canonical text model — the concatenation of text
 *  nodes ONLY. Chromium's Range.toString() (used for live caret offsets) does
 *  NOT include `<br>` elements NOR div boundaries, and the SHOW_TEXT
 *  TreeWalker (used to map offsets back onto DOM positions) agrees with it.
 *  The previous version translated `<br>` into `\n` — desyncing the three
 *  views by one char per `<br>` and causing off-by-N errors: after the first
 *  mention (which inserts a `<br>`), every later `@图X` was anchored at the
 *  wrong offset and the chip ended up appended at the END of the whole prompt
 *  on a new line, no matter where the user typed `@`. */
export function htmlToText(html: string): string {
  if (!html) return ''
  let s = html.replace(/<br\s*\/?>/gi, '')
  // Skill 胶囊是整段 token 的可视化：span 被删一半/属性丢失时，
  // 里面的 `◆ 名字` 裸文本不能残留，统一还原成 token 再做完整性收尾。
  s = s.replace(/<span[^>]*data-skill="([^"]*)"[^>]*>.*?<\/span>/gi, (_m, n) => `⟦skill:${n}⟧`)
  s = s.replace(/<span[^>]*class="[^"]*skill-chip[^"]*"[^>]*>.*?<\/span>/gi, '')
  s = s.replace(/<[^>]+>/g, '')
  s = s.replace(/&nbsp;/g, ' ').replace(/&amp;/g, '&').replace(/&lt;/g, '<').replace(/&gt;/g, '>')
  // 残缺 token 收尾：胶囊只删了一半时会剩下 `⟦skill:名字` / `名字⟧` 碎片，
  // 统一清除，保证文本模型里要么是完整 token、要么什么都没有。
  s = s.replace(/⟦skill:[^\]⟧\r\n]*$/g, '')
  s = s.replace(/^[^\[⟦\r\n]*⟧/g, '')
  s = s.replace(/⟦skill:[^\]⟧\r\n]*[\r\n]/g, '')
  return s
}

/** Map an absolute character index (canonical text model: text nodes only) to
 *  the DOM text node + offset containing it. Falls back to the last text
 *  node's end for indices at/after the total length. */
function findTextPos(el: HTMLElement, abs: number): { node: Text; offset: number } | null {
  const walker = document.createTreeWalker(el, NodeFilter.SHOW_TEXT)
  let acc = 0
  let node: Node | null
  while ((node = walker.nextNode())) {
    const tn = node as Text
    const len = tn.data.length
    if (acc + len >= abs) return { node: tn, offset: abs - acc }
    acc += len
  }
  // abs may equal total length — fall back to last text node, end offset.
  const all: Text[] = []
  const w2 = document.createTreeWalker(el, NodeFilter.SHOW_TEXT)
  let n: Node | null
  while ((n = w2.nextNode())) all.push(n as Text)
  const last = all[all.length - 1]
  if (last) return { node: last, offset: last.data.length }
  return null
}

/** The text node that follows `node` in document order, or null. */
function nextTextNodeAfter(el: HTMLElement, node: Node): Text | null {
  const walker = document.createTreeWalker(el, NodeFilter.SHOW_TEXT)
  let n: Node | null
  let seen = false
  while ((n = walker.nextNode())) {
    if (seen) return n as Text
    if (n === node) seen = true
  }
  return null
}

/** Live caret offset in the canonical text model, or -1 when unavailable. */
function caretTextOffset(el: HTMLElement): number {
  const sel = window.getSelection()
  if (!sel || sel.rangeCount === 0) return -1
  try {
    const range = sel.getRangeAt(0)
    if (!el.contains(range.startContainer)) return -1
    const pre = document.createRange()
    pre.selectNodeContents(el)
    pre.setEnd(range.startContainer, range.startOffset)
    return pre.toString().length
  } catch {
    return -1
  }
}

/** True when the '@' at character index `idx` belongs to an EXISTING mention
 *  chip (an old @图N), not to a fresh query the user is typing. */
function isMentionChipAt(el: HTMLElement, text: string, idx: number): boolean {
  if (text[idx] !== '@') return false
  let node = findTextPos(el, idx)
  if (!node) return false
  // idx may sit on a node boundary (end of a text node); the '@' then starts
  // the NEXT text node (e.g. a chip that begins a new line).
  if (node.offset >= node.node.data.length) {
    const next = nextTextNodeAfter(el, node.node)
    if (!next) return false
    node = { node: next, offset: 0 }
  }
  if (node.node.data[node.offset] !== '@') return false
  const parent = node.node.parentElement
  return !!parent && !!parent.closest('.mention')
}

// print-frame 边长：实测所在列的可用宽×行高后取 min（封顶 560px，防超大屏糊成一片）。
// 不再写死 400：窗口拉大拉小，正方形永远取“当前空间里最大的合法边长”。
function usePrintSize(colRef: React.RefObject<HTMLDivElement | null>): number {
  const [size, setSize] = useState(400)
  useLayoutEffect(() => {
    const el = colRef.current
    if (!el) return
    const measure = () => {
      const row = el.parentElement
      const head = el.parentElement?.querySelector(':scope > .eyebrow')
      const rowH = row?.clientHeight ?? 400
      const headH = head instanceof HTMLElement ? head.offsetHeight : 28
      const w = el.clientWidth
      // 行高 - header - 列内 gap ≈ 正方形可用高度；列宽即宽度上限
      const next = Math.max(200, Math.min(560, w, rowH - headH - 2))
      setSize((prev) => (Math.abs(prev - next) < 1 ? prev : next))
    }
    measure()
    const ro = new ResizeObserver(measure)
    if (el.parentElement) ro.observe(el.parentElement)
    ro.observe(el)
    window.addEventListener('resize', measure)
    return () => {
      ro.disconnect()
      window.removeEventListener('resize', measure)
    }
  }, [colRef])
  return size
}

// URL 回包下载：60s 超时 + 50MB 体积上限。
// 之前是裸 net.request：中转站返回超大/假死 URL 会一直挂着不回来，
// 内存里攒 Buffer 也可能把主进程撑爆。超限直接报 io 错，不写盘、不进历史。
const URL_DOWNLOAD_TIMEOUT_MS = 60_000
const URL_DOWNLOAD_MAX_BYTES = 50 * 1024 * 1024

async function downloadIfUrl(url: string): Promise<{ data: Buffer; mime: string } | { ok: false }> {
  try {
    const { net } = await import('electron')
    return await new Promise((resolve) => {
      let settled = false
      const done = (v: { data: Buffer; mime: string } | { ok: false }) => {
        if (settled) return
        settled = true
        clearTimeout(timer)
        resolve(v)
      }
      const timer = setTimeout(() => {
        try { (req as unknown as { abort?: () => void }).abort?.() } catch { /* ignore */ }
        done({ ok: false })
      }, URL_DOWNLOAD_TIMEOUT_MS)
      const req = net.request({ url, method: 'GET' })
      const chunks: Buffer[] = []
      let received = 0
      let mime = 'image/png'
      req.on('response', (resp) => {
        // Content-Length 先验：超限直接掐掉，连 body 都不收
        const cl = resp.headers['content-length']
        const declared = Array.isArray(cl) ? cl[0] : cl
        if (typeof declared === 'string' && /^\d+$/.test(declared.trim())) {
          if (Number(declared) > URL_DOWNLOAD_MAX_BYTES) {
            try { (req as unknown as { abort?: () => void }).abort?.() } catch { /* ignore */ }
            done({ ok: false })
            return
          }
        }
        const ct = resp.headers['content-type']
        if (typeof ct === 'string') mime = ct.split(';')[0] || mime
        resp.on('data', (c) => {
          received += (c as Buffer).length
          if (received > URL_DOWNLOAD_MAX_BYTES) {
            try { (req as unknown as { abort?: () => void }).abort?.() } catch { /* ignore */ }
            done({ ok: false })
            return
          }
          chunks.push(c as Buffer)
        })
        resp.on('end', () => done({ data: Buffer.concat(chunks), mime }))
        resp.on('error', () => done({ ok: false } as never))
      })
      req.on('error', () => done({ ok: false } as never))
      req.end()
    })
  } catch {
    return { ok: false } as never
  }
}

import axios, { type AxiosError, type AxiosResponse } from 'axios'
import type { GenerateRequest, GenerateResult } from '../shared/types'
import { loadSettings } from './settings'
import { activePreset } from './presets'

const CONTROLLERS = new Map<string, AbortController>()
// 串行：所有生成请求经 generateQueue.enqueue FIFO 排队（与 CLI / MCP 三端同队列语义），
// 按下顺序逐个执行，不再 busy 拒绝。三端共用 shared/headless-schema.ts 组装请求体。

const MAX_PAYLOAD_BYTES = 20 * 1024 * 1024

function bodySnippet(s: string): string {
  return s.length > 2000 ? s.slice(0, 2000) + '…' : s
}

export async function generate(req: GenerateRequest): Promise<GenerateResult> {
  const settings = loadSettings()
  const preset = activePreset(settings)
  if (!preset.baseUrl) return { ok: false, code: 'config', message: '当前预设未配置 API Base URL，请到设置中编辑预设' }
  if (!preset.apiKey) return { ok: false, code: 'config', message: '当前预设未配置 API Key，请到设置中编辑预设' }

  // Payload size check (base64 strings are the heavy part).
  const payloadBytes = estimatePayloadBytes(req.images)
  if (payloadBytes > MAX_PAYLOAD_BYTES) {
    return { ok: false, code: 'payload', message: `参考图总体积 ${(payloadBytes / 1024 / 1024).toFixed(1)}MB 过大（建议 ≤ 20MB）` }
  }

  const presetLike: PresetLike = {
    model: preset.model,
    imageFieldName: preset.imageFieldName,
    sizeMode: preset.sizeMode,
    sendAspectRatio: preset.sendAspectRatio,
  }
  const body = buildImageBody(
    {
      prompt: req.promptResolved,
      aspect: req.params.aspect,
      resolution: req.params.resolution,
      quality: req.params.quality,
      count: req.params.count,
      images: req.images,
    },
    presetLike,
    req.promptResolved,
  )

  const url = `${preset.baseUrl}/v1/images/generations`

  console.log('[generate] enqueue', req.requestId, 'queueDepth=', generateQueue.pending())
  return generateQueue.enqueue(async (): Promise<GenerateResult> => {
    console.log('[generate] dequeue start', req.requestId)
    const ctrl = new AbortController()
    CONTROLLERS.set(req.requestId, ctrl)
    try {
      const resp: AxiosResponse = await axios.post(url, body, {
        headers: {
          Authorization: `Bearer ${preset.apiKey}`,
          'Content-Type': 'application/json',
        },
        timeout: settings.timeoutMs,
        maxBodyLength: Infinity,
        maxContentLength: Infinity,
        signal: ctrl.signal,
        responseType: 'json',
      })
      const r = await normalizeResponse(req, resp.data)
      console.log('[generate] done', req.requestId, r.ok ? 'ok' : r.code)
      return r
    } catch (e) {
      const ax = e as AxiosError
      if (ax.name === 'CanceledError' || ax.code === 'ERR_CANCELED') {
        console.log('[generate] canceled', req.requestId)
        return { ok: false, code: 'canceled', message: '已取消' }
      }
      if (ax.code === 'ECONNABORTED') {
        console.log('[generate] timeout', req.requestId)
        return { ok: false, code: 'timeout', message: `请求超时（${settings.timeoutMs}ms）` }
      }
      if (ax.response) {
        const status = ax.response.status
        const data = ax.response.data
        const bodyStr = typeof data === 'string' ? data : (() => { try { return JSON.stringify(data) } catch { return String(data) } })()
        console.log('[generate] http', req.requestId, status, bodySnippet(bodyStr).slice(0, 200))
        return { ok: false, code: 'http', status, message: `HTTP ${status}`, body: bodySnippet(bodyStr) }
      }
      console.log('[generate] network error', req.requestId, ax.message)
      return { ok: false, code: 'network', message: ax.message || '网络错误' }
    } finally {
      CONTROLLERS.delete(req.requestId)
    }
  })
}

export function cancel(requestId: string): void {
  const c = CONTROLLERS.get(requestId)
  if (c) {
    c.abort()
    CONTROLLERS.delete(requestId)
  }
  console.log('[generate] cancel requested', requestId)
}
