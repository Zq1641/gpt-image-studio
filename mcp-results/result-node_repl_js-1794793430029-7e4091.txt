// URL 回包下载：60s 超时 + 50MB 体积上限。
// 之前是裸 net.request：中转站返回超大/假死 URL 会一直挂着不回来，
// 内存里攒 Buffer 也可能把主进程撑爆。超限直接报 io 错，不写盘、不进历史。
const URL_DOWNLOAD_TIMEOUT_MS = 60_000
const URL_DOWNLOAD_MAX_BYTES = 50 * 1024 * 1024

async function downloadIfUrl(url: string): Promise<{ data: Buffer; mime: string } | { ok: false }> {
  try {
    const { net } = await import('electron')
    return await new Promise((resolve) => {
      let settled = false
      const done = (v: { data: Buffer; mime: string } | { ok: false }) => {
        if (settled) return
        settled = true
        clearTimeout(timer)
        resolve(v)
      }
      const timer = setTimeout(() => {
        try { (req as unknown as { abort?: () => void }).abort?.() } catch { /* ignore */ }
        done({ ok: false })
      }, URL_DOWNLOAD_TIMEOUT_MS)
      const req = net.request({ url, method: 'GET' })
      const chunks: Buffer[] = []
      let received = 0
      let mime = 'image/png'
      req.on('response', (resp) => {
        // Content-Length 先验：超限直接掐掉，连 body 都不收
        const cl = resp.headers['content-length']
        const declared = Array.isArray(cl) ? cl[0] : cl
        if (typeof declared === 'string' && /^\d+$/.test(declared.trim())) {
          if (Number(declared) > URL_DOWNLOAD_MAX_BYTES) {
            try { (req as unknown as { abort?: () => void }).abort?.() } catch { /* ignore */ }
            done({ ok: false })
            return
          }
        }
        const ct = resp.headers['content-type']
        if (typeof ct === 'string') mime = ct.split(';')[0] || mime
        resp.on('data', (c) => {
          received += (c as Buffer).length
          if (received > URL_DOWNLOAD_MAX_BYTES) {
            try { (req as unknown as { abort?: () => void }).abort?.() } catch { /* ignore */ }
            done({ ok: false })
            return
          }
          chunks.push(c as Buffer)
        })
        resp.on('end', () => done({ data: Buffer.concat(chunks), mime }))
        resp.on('error', () => done({ ok: false } as never))
      })
      req.on('error', () => done({ ok: false } as never))
      req.end()
    })
  } catch {
    return { ok: false } as never
  }
}

export async function normalizeResponse(req: GenerateRequest, raw: unknown): Promise<GenerateResult> {
  const images = extractImages(raw)
  if (images.length === 0) {
    const bodyStr = typeof raw === 'string' ? raw : (() => { try { return JSON.stringify(raw) } catch { return String(raw) } })()
    return { ok: false, code: 'parse', message: '未识别的响应格式', body: bodyStr.slice(0, 2000) }
  }

  // If the proxy returned FEWER images than we asked for, the upstream model
  // probably capped `n` silently (or the proxy dropped a plate while
  // re-wrapping the response). Log it loudly so a developer running with
  // DevTools open sees what really came back — and include it in the user-
  // facing message so the discrepancy isn't hidden.
  const requested = req.params.count
  if (images.length < requested) {
    console.warn(`[normalize] requested ${requested} image(s), proxy returned ${images.length}`)
  }

  const entry: HistoryEntry = {
    id: id(),
    createdAt: Date.now(),
    presetName: activePreset(loadSettings()).name,
    model: activePreset(loadSettings()).model,
    promptRaw: req.promptRaw,
    promptResolved: req.promptResolved,
    params: req.params,
    size: sizeString(req.params.aspect, req.params.resolution, 'computed'),
    refImageFiles: [],
    // 参考图快照随历史持久化，回填时可直接恢复（老条目没有则为空）。
    // 注意体积：快照是 JPEG dataUrl，history.json 会随条目线性变大，
    // 瘦身请用 Cache 弹窗的“参考图快照”清理（只剥快照、不动历史）。
    refImages: Array.isArray(req.refImages) ? req.refImages : [],
    images: [],
  }

  const dir = imagesDir()
  const saved: RawImage[] = []
  for (let i = 0; i < images.length; i++) {
    const img = images[i]
    if (!img) continue
    const name = `${safeId(req.requestId)}-${i}.png`
    try {
      if (img.kind === 'b64') {
        const bytes = Buffer.from(img.data, 'base64')
        writeFileSync(join(dir, name), bytes)
        saved.push({ file: name, url: `img://${name}`, mime: 'image/png', bytes: bytes.length })
        console.log('[localFile] saved', name, bytes.length + 'B', `img://${name}`);
      } else {
        const got = await downloadIfUrl(img.url)
        if ('ok' in got) {
          return { ok: false, code: 'io', message: '下载生成图片失败（超时或文件过大，限 60s / 50MB）' }
        }
        writeFileSync(join(dir, name), got.data)
        saved.push({ file: name, url: `img://${name}`, mime: 'image/png', bytes: got.data.length })
      }
    } catch (e) {
      return { ok: false, code: 'io', message: `写入图片失败：${(e as Error).message}` }
    }
  }

  entry.images = saved.map((s) => ({ file: s.file, url: s.url }))
  console.log('[job] job_id=', req.requestId, 'saved', saved.length + '/' + images.length, 'size=', req.params.aspect, req.params.resolution);
  console.log('[file] protocol img:// files=', saved.map((s) => s.file).slice(0, 4).join(','));
  return { ok: true, entry }
}

type Extracted = { kind: 'b64'; data: string } | { kind: 'url'; url: string }

// Pull every image we can find out of an arbitrary response object.
//
// Proxies in the wild wrap OpenAI-style responses in many shapes. We used
// to only handle the canonical `{ data: [{ b64_json }, ...] }` and top-level
// `b64_json` / `url` fallbacks. That dropped images silently when a proxy
// used e.g. `{ data: { b64_json: ["...", "..."] } }` (data is an OBJECT,
// not an array) or wrapped urls differently, so a 2-image request would
// surface only 1 — and the user saw "API charged for 2 but UI shows 1".
//
// This walker accepts the canonical format plus the common wrappers:
//   - canonical array:   { data: [{ b64_json } | { url }, ...] }
//   - alt array keys:    { images: [...] }, { output: [...] }, { result: [...] }
//   - data-as-object:    { data: { b64_json: "..." | [...] } }
//                        { data: { url: "..." | [...] } }
//                        { data: { images: [...] } }
//   - top-level scalar:  { b64_json: "..." }, { url: "..." }
//   - top-level array:   { b64_json: ["...", "..."] }, { url: ["...", "..."] }
function extractImages(raw: unknown): Extracted[] {
  if (!raw || typeof raw !== 'object') return []
  const out: Extracted[] = []
  const r = raw as Record<string, unknown>

  const pushStrings = (v: unknown, kind: 'b64' | 'url') => {
    if (typeof v === 'string' && v.length > 0) {
      out.push(kind === 'b64' ? { kind: 'b64', data: v } : { kind: 'url', url: v })
    } else if (Array.isArray(v)) {
      for (const x of v) {
        if (typeof x === 'string' && x.length > 0) {
          out.push(kind === 'b64' ? { kind: 'b64', data: x } : { kind: 'url', url: x })
        }
      }
    }
  }

  const arrayKeys = ['data', 'images', 'output', 'result']
  for (const key of arrayKeys) {
    const v = r[key]
    if (Array.isArray(v)) {
      for (const item of v) {
        if (!item || typeof item !== 'object') continue
        const it = item as Record<string, unknown>
        if (typeof it.b64_json === 'string') out.push({ kind: 'b64', data: it.b64_json })
        else if (typeof it.url === 'string') out.push({ kind: 'url', url: it.url })
      }
    } else if (v && typeof v === 'object') {
      // data-as-object: { data: { b64_json: ... | [...] } } etc.
      const inner = v as Record<string, unknown>
      pushStrings(inner.b64_json, 'b64')
      pushStrings(inner.url, 'url')
      if (Array.isArray(inner.images)) {
        for (const item of inner.images) {
          if (!item || typeof item !== 'object') continue
          const it = item as Record<string, unknown>
          if (typeof it.b64_json === 'string') out.push({ kind: 'b64', data: it.b64_json })
          else if (typeof it.url === 'string') out.push({ kind: 'url', url: it.url })
        }
      }
    }
  }

  // Top-level scalar / array fallbacks.
  pushStrings(r.b64_json, 'b64')
  pushStrings(r.url, 'url')

  // Dedup by content (some proxies duplicate b64_json at top-level AND
  // inside data — we'd otherwise write the same plate twice).
  const seen = new Set<string>()
  return out.filter((e) => {
    const key = e.kind === 'b64' ? `b:${e.data.slice(0, 64)}:${e.data.length}` : `u:${e.url}`
    if (seen.has(key)) return false
    seen.add(key)
    return true
  })
}

// 下载类 IPC 只接受「images/ 目录下的单个 png 文件名」，不接受任何路径。
// 白名单 ^[A-Za-z0-9_.-]+\.png$ 直接挡掉 ../、绝对路径、分隔符和空字节；
// 再加长度上限，避免超长文件名拖垮文件系统 / 对话框。
const SAFE_IMAGE_FILENAME = /^[A-Za-z0-9_.-]+\.png$/
const SAFE_IMAGE_FILENAME_MAX = 200

export function isSafeImageFilename(name: unknown): name is string {
  return (
    typeof name === 'string' &&
    name.length > 0 &&
    name.length <= SAFE_IMAGE_FILENAME_MAX &&
    SAFE_IMAGE_FILENAME.test(name) &&
    !name.includes('..')
  )
}

// 把通过白名单的文件名解析成 images/ 下的绝对路径，并再做一次 startsWith 兜底，
// 确保最终读写的路径一定落在 imagesDir 内（防 join 之后的路径逃逸）。
function safeImagePath(name: string): string | null {
  if (!isSafeImageFilename(name)) return null
  const dir = imagesDir()
  const abs = resolve(join(dir, name))
  if (abs !== join(dir, name) || !abs.startsWith(resolve(dir) + sep)) return null
  return abs
}

function readIndex(): HistoryEntry[] {
  const p = indexPath()
  if (!existsSync(p)) return []
  try {
    const raw = readFileSync(p, 'utf-8')
    const arr = JSON.parse(raw) as HistoryEntry[]
    if (!Array.isArray(arr)) return []
    return arr
  } catch {
    // history.json 损坏时：把坏文件改名为 history.corrupt-<时间戳>.json 留证，
    // 返回空列表让应用照常启动。只备份、不删除——用户数据一条都不丢，
    // 修好 JSON 后改回文件名即可恢复。
    try {
      const stamp = new Date().toISOString().replace(/[:.]/g, '-').slice(0, 19)
      renameSync(p, join(userDataDir(), `history.corrupt-${stamp}.json`))
    } catch { /* 备份失败也无妨，下次启动再试 */ }
    return []
  }
}

function writeIndex(arr: HistoryEntry[]): void {
  const dir = userDataDir()
  if (!existsSync(dir)) mkdirSync(dir, { recursive: true })
  // 原子写：先写 tmp 再 rename，避免进程崩溃时 history.json 写一半变坏。
  // tmp 与目标同目录，保证 rename 是同文件系统原子操作。
  const p = indexPath()
  const tmp = `${p}.tmp`
  writeFileSync(tmp, JSON.stringify(arr, null, 2), 'utf-8')
  renameSync(tmp, p)
}

export function listHistory(): HistoryEntry[] {
  return readIndex().sort((a, b) => b.createdAt - a.createdAt)
}

export function appendHistory(entry: HistoryEntry): HistoryEntry[] {
  const arr = readIndex()
  arr.push(entry)
  writeIndex(arr)
  return listHistory()
}

export function deleteHistory(id: string): HistoryEntry[] {
  const arr = readIndex()
  const target = arr.find((e) => e.id === id)
  const rest = arr.filter((e) => e.id !== id)
  writeIndex(rest)
  // 单条删除同时回收它独占的生成图文件：先收集剩余条目仍在引用的文件名，
  // 只删除不再被任何条目引用的文件，避免误删多条历史共享同名 plate 的极端情况。
  try {
    const stillUsed = new Set<string>()
    for (const e of rest) {
      for (const img of e.images ?? []) {
        if (img?.file) stillUsed.add(img.file)
      }
    }
    const dir = imagesDir()
    for (const img of target?.images ?? []) {
      const f = img?.file
      if (!f || stillUsed.has(f)) continue
      if (f.includes('..') || f.includes('/') || f.includes('\\')) continue
      const abs = join(dir, f)
      if (existsSync(abs)) {
        try { unlinkSync(abs) } catch { /* ignore */ }
      }
    }
  } catch { /* GC 失败不影响删除本身 */ }
  return listHistory()
}

export function clearHistory(): HistoryEntry[] {
  const dir = imagesDir()

    for (const f of readdirSync(dir)) {
      try { unlinkSync(join(dir, f)) } catch { /* ignore */ }
    }
  } catch { /* dir 可能不存在，没问题 */ }

  return []
}

// ---------------- 缓存统计 + 按项清理 ----------------
// 体积口径：
//   - imageBytes：images/ 目录下全部生成图文件的实际字节
//   - historyBytes：history.json 文件体积（含内嵌的参考图快照）
//   - refBytes：历史条目里 refImages[].dataUrl 的 base64 估算字节（history.json 的子集，
//     单独列出来是为了让用户看清“参考图快照占了多少”，清理时可单独剥离）
//   - totalBytes = imageBytes + historyBytes（用户磁盘上真实占用的合计数）

function approxDataUrlBytes(dataUrl: string): number {
  const i = dataUrl.indexOf(',')
  const b64 = i >= 0 ? dataUrl.slice(i + 1) : dataUrl
  return Math.floor(b64.length * 0.75)
}

export function getCacheStats(): CacheStats {
  const dir = imagesDir()
  let imageCount = 0
  let imageBytes = 0
  try {
    for (const f of readdirSync(dir)) {
      try {
        const st = statSync(join(dir, f))
        if (st.isFile()) {
          imageCount += 1
          imageBytes += st.size
        }
      } catch { /* ignore */ }
    }
  } catch { /* dir 可能不存在 */ }

  let historyBytes = 0
  try {
    const st = statSync(indexPath())
    if (st.isFile()) historyBytes = st.size
  } catch { /* ignore */ }

  const entries = readIndex()
  let refCount = 0
  let refBytes = 0
  for (const e of entries) {
    const refs = Array.isArray(e.refImages) ? e.refImages : []
    for (const r of refs) {
      refCount += 1
      if (typeof r?.dataUrl === 'string') refBytes += approxDataUrlBytes(r.dataUrl)
    }
  }

  return {
    imageCount,
    imageBytes,
    historyCount: entries.length,
    historyBytes,
    refCount,
    refBytes,
    totalBytes: imageBytes + historyBytes,
    userDataDir: userDataDir(),
    imagesDir: dir,
  }
}

// 改名迁移：appId 从 com.pony.gptimage2 → com.pony.gptimagestudio，
// productName 从 "GPT Image 2" → "GPT Image" 后，userData 目录会变化，
// 旧的历史/设置/图片全在老目录里。首次启动时把老目录内容拷贝过来
// （只拷贝、不删除老目录、不覆盖新目录已有文件），老用户升级不断档。
//
// 老目录有两种可能（都要兼容）：
//   1. 打包版旧应用：`…/Application Support/GPT Image 2`（按旧 productName 推导）
//   2. 开发版跑出来的：`…/Application Support/gpt-image-2`（按旧 package.json name 推导）
// 候选按优先级排序，命中第一个含数据的即用，不逐个合并（防两个老目录各一份导致混拷）。
function migrateLegacyUserData(): void {
  try {
    const next = app.getPath('userData')
    const appData = app.getPath('appData')
    const candidates = [
      join(appData, 'GPT Image 2'),
      join(appData, 'gpt-image-2'),
    ].filter((p) => p !== next)
    const legacy = candidates.find((p) => {
      try {
        if (!existsSync(p) || !statSync(p).isDirectory()) return false
        // 含任一有效数据才算命中：空的 Electron 残留目录不算
        if (existsSync(join(p, 'settings.json'))) return true
        if (existsSync(join(p, 'history.json'))) return true
        try { return readdirSync(join(p, 'images')).length > 0 } catch { return false }
      } catch { return false }
    })
    if (!legacy) return
    if (!existsSync(next)) mkdirSync(next, { recursive: true })
    // 只有新目录完全没数据时才迁：避免把旧快照覆盖掉新用户已用的配置。
    // 新目录已有 settings.json/history.json/images 任一 → 视为已迁移或新用户，跳过。
    const hasData =
      existsSync(join(next, 'settings.json')) ||
      existsSync(join(next, 'history.json')) ||
      (() => {
        try { return readdirSync(join(next, 'images')).length > 0 } catch { return false }
      })()
    if (hasData) return
    let copied = 0
    let skipped = 0
    for (const name of ['settings.json', 'history.json']) {
      const src = join(legacy, name)
      const dest = join(next, name)
      try {
        if (!existsSync(src) || !statSync(src).isFile() || existsSync(dest)) { skipped += 1; continue }
        copyFileSync(src, dest)
        copied += 1
      } catch (e) {
        console.error(`[migrate] copy failed: ${name}`, e)
      }
    }
    // images 目录逐文件拷贝（只拷缺失、不覆盖）：cpSync 整目录在 dest 已存在时
    // 行为不一致，逐文件最稳，也能逐个统计。
    try {
      const srcDir = join(legacy, 'images')
      if (existsSync(srcDir) && statSync(srcDir).isDirectory()) {
        const destDir = join(next, 'images')
        if (!existsSync(destDir)) mkdirSync(destDir, { recursive: true })
        for (const f of readdirSync(srcDir)) {
          try {
            const src = join(srcDir, f)
            const dest = join(destDir, f)
            if (!statSync(src).isFile() || existsSync(dest)) { skipped += 1; continue }
            copyFileSync(src, dest)
            copied += 1
          } catch { skipped += 1 }
        }
      }
    } catch (e) {
      console.error('[migrate] copy failed: images', e)
    }
    console.log(`[migrate] legacy userData copied: ${legacy} -> ${next} (copied=${copied} skipped=${skipped}, legacy kept)`)
  } catch (e) {
    // 迁移失败不阻塞启动：用户最多是旧历史没带过来，老目录原样保留可手动拷。
    console.error('[migrate] legacy userData migration failed', e)
  }
}

// API Key 只在「落盘」这一刻加密、「读取」这一刻解密：
// macOS 下 safeStorage 底层走系统钥匙串，盘上不再出现明文 Key。

export function loadSettings(): AppSettings {
  const p = settingsPath()
  if (!existsSync(p)) return ensureActive({ ...DEFAULT_SETTINGS })
  try {
    const raw = readFileSync(p, 'utf-8')
    const parsed = JSON.parse(raw) as Partial<AppSettings> & {
      baseUrl?: string; apiKey?: string; model?: string

      sizeMode?: 'computed' | 'square' | 'omit'
      sendAspectRatio?: boolean

    if (parsed.presets == null && (parsed.baseUrl || parsed.apiKey)) {
      const legacy: Preset = normalizePreset({
        id: 'default',
        name: '默认预设',
        baseUrl: parsed.baseUrl ?? '',
        apiKey: «redacted 13 chars» ?? '',
        pricePerImage: 0,
        model: parsed.model || 'gpt-image-2',
        imageFieldName: (parsed.imageFieldName ?? 'image') as Preset['imageFieldName'],
        sizeMode: (parsed.sizeMode ?? 'computed') as Preset['sizeMode'],
        sendAspectRatio: parsed.sendAspectRatio ?? true,
      })
      const ensured = ensureActive({ ...DEFAULT_SETTINGS, presets: [legacy], activePresetId: legacy.id })
      // 老明文盘 → 立刻回写成加密盘，避免明文继续留在磁盘上
      try { writeSettings(ensured) } catch { /* ignore */ }
      return ensured
    }
    // 兼容旧版 settings.json：normalizeModels 缺字段/脏数据时都补默认列表
    const models = normalizeModels(parsed.models)
    const merged: AppSettings = { ...DEFAULT_SETTINGS, ...(parsed as Partial<AppSettings>), models }
    const ensured = decryptPresets(ensureActive(merged))
    // 关键修复：若旧盘上 models 字段缺失/非法，ensureActive 只在内存里补了默认，
    // 不写盘会导致“每次加载看起来有 4 个、但删除永远删不掉”的幽灵列表。
    // 这里一次性回填写盘，后续增删就有稳定的基准了。
    // 顺带把「历史明文 Key」升级为加密盘（写回时统一加密）。
    if (!Array.isArray(parsed.models) || (parsed.presets ?? []).some((p) => p?.apiKey && !isEncryptedApiKey(p.apiKey))) {
      try { writeSettings(ensured) } catch { /* ignore */ }
    }
    return ensured
  } catch {
    return ensureActive({ ...DEFAULT_SETTINGS })
  }
}

export function saveSettings(patch: Partial<AppSettings>): AppSettings {
  const cur = loadSettings()
  // saveSettings 是全量 patch 合并：若调用方没带 models，必须保留当前盘上的，
  // 否则一次“只改 timeout”之类的保存就会把用户删过的模型又恢复出来。
  const next = ensureActive({
    ...cur,
    ...patch,
    models: patch.models === undefined ? cur.models : normalizeModels(patch.models),
  })
  writeSettings(next)
  return next
}

export function addSkill(input: { name: string; content: string; sourceName?: string }): AppSettings {
  const cur = loadSettings()
  const name = normalizeSkillName(input.name)
  const content = typeof input.content === 'string' ? input.content : ''
  if (!name || !content.trim() || content.length > SKILL_CONTENT_MAX) return cur
  const skills = normalizeSkillList(cur.skills)
  const sourceName = typeof input.sourceName === 'string' ? input.sourceName : ''
  const at = skills.findIndex((s) => s.name === name)
  if (at >= 0) {
    const prev = skills[at]!
    skills[at] = { ...prev, content, sourceName, charCount: content.length }
  } else {
    skills.push({ id: newSkillId(), name, content, sourceName, charCount: content.length, createdAt: Date.now() })
  }
  const next: AppSettings = ensureActive({ ...cur, skills })
  writeSettings(next)
  return next
}

export function deleteSkill(id: string): AppSettings {
  const cur = loadSettings()
  const skills = normalizeSkillList(cur.skills)
  if (!skills.some((s) => s.id === id)) return cur
  const next: AppSettings = ensureActive({ ...cur, skills: skills.filter((s) => s.id !== id) })
  writeSettings(next)
  return next
}

function normalizeModels(input: unknown): string[] {
  const list = Array.isArray(input) ? input : []
  const cleaned = list
    .filter((x): x is string => typeof x === 'string')
    .map((x) => x.trim())
    .filter((x) => x.length > 0 && x.length <= 120)
  const deduped = Array.from(new Set(cleaned))
  return deduped.length > 0 ? deduped : [...DEFAULT_MODELS]
}

export function addModel(name: string): AppSettings {
  const cur = loadSettings()
  const trimmed = (name ?? '').trim()
  if (!trimmed || trimmed.length > 120) return cur
  const models = normalizeModels(cur.models)
  if (!models.includes(trimmed)) models.push(trimmed)
  else return cur // 已存在：直接返回，避免无意义写盘触发 UI 抖动
  const next: AppSettings = ensureActive({ ...cur, models })
  writeSettings(next)
  return next
}

export function deleteModel(name: string): AppSettings {
  const cur = loadSettings()
  const trimmed = (name ?? '').trim()
  if (!trimmed) return cur
  const before = normalizeModels(cur.models)
  const models = before.filter((m) => m !== trimmed)
  // 至少保留一个模型；要删的不存在时直接返回，避免无意义写盘
  if (models.length === 0 || models.length === before.length) return cur
  const next: AppSettings = ensureActive({ ...cur, models })
  // 若当前激活预设用的模型被删了，不动预设（预设允许自定义模型名），只收窄全局列表；
  // 侧边栏会以虚线 custom chip 展示该预设模型，保证大字与 chips 对得上。
  writeSettings(next)
  return next
}

function writeSettings(s: AppSettings): void {
  const dir = app.getPath('userData')
  if (!existsSync(dir)) mkdirSync(dir, { recursive: true })
  // 落盘前把每个预设的 API Key 加密（内存里的 s 不变，返回给渲染进程仍是明文）。
  const onDisk = encryptPresets(s)
  const p = settingsPath()
  // 原子写：先写同目录 tmp 再 rename，避免进程崩溃时 settings.json 写一半变坏。
  const tmp = `${p}.tmp`
  writeFileSync(tmp, JSON.stringify(onDisk, null, 2), { encoding: 'utf-8', mode: 0o600 })
  // 收紧权限：即使进程 umask 宽松，含 API Key 的文件也只允许当前用户读写。
  try { chmodSync(tmp, 0o600) } catch { /* ignore */ }
  renameSync(tmp, p)
  try { chmodSync(p, 0o600) } catch { /* ignore */ }
}

function usePrintSize(colRef: React.RefObject<HTMLDivElement | null>): number {
  const [size, setSize] = useState(400)
  useLayoutEffect(() => {
    const el = colRef.current
    if (!el) return
    const measure = () => {
      const row = el.parentElement
      const head = el.parentElement?.querySelector(':scope > .eyebrow')
      const rowH = row?.clientHeight ?? 400
      const headH = head instanceof HTMLElement ? head.offsetHeight : 28
      const w = el.clientWidth
      // 行高 - header - 列内 gap ≈ 正方形可用高度；列宽即宽度上限
      const next = Math.max(200, Math.min(560, w, rowH - headH - 2))
      setSize((prev) => (Math.abs(prev - next) < 1 ? prev : next))
    }
    measure()
    const ro = new ResizeObserver(measure)
    if (el.parentElement) ro.observe(el.parentElement)
    ro.observe(el)
    window.addEventListener('resize', measure)
    return () => {
      ro.disconnect()
      window.removeEventListener('resize', measure)
    }
  }, [colRef])
  return size

function reducer(state: ComposerState, action: Action): ComposerState {
  switch (action.type) {
    case 'set-prompt': return { ...state, prompt: action.value }
    case 'set-param': return { ...state, params: { ...state.params, [action.key]: action.value } }
    case 'apply-template': {
      const t = TEMPLATES.find((x) => x.id === action.id)
      return t ? { ...state, prompt: t.text } : state
    }
    case 'set-refs': return { ...state, refImages: action.refs }
    case 'set-status':
      return {
        ...state,
        status: action.status,
        error: action.error,
        activeRequestId: action.status === 'loading' ? action.requestId : undefined,
      }
    case 'set-results':
      return { ...state, results: action.results, previewEntry: action.entry, activeRequestId: undefined }
    case 'set-history': return { ...state, history: action.history }
    case 'set-settings': return { ...state, settings: action.settings }
    case 'set-preview': return { ...state, previewEntry: action.entry, results: action.entry?.images ?? [] }
    default: return state
  }
}
