/** 把纯文本里的 `@图N` (N 是数字) 替换成高亮 HTML，
 *  把 `⟦skill:NAME⟧` 替换成红色 Skill 胶囊。其它字符 escaped。
 *  注意顺序：先 escape → skill token（含 ⟦⟧，escape 不动它们）→ @图N。 */
export function mentionizeMarkup(plain: string): string {
  if (!plain) return ''
  const escaped = plain
    .replace(/&/g, '&amp;')
    .replace(/</g, '&lt;')
    .replace(/>/g, '&gt;')
  // Skill 胶囊：只显示名字（全文生成时后台拼入）。
  // 名字已在入库时去掉 HTML 特殊字符（见 normalizeSkillName），直接匹配即可。
  const withSkills = escaped.replace(
    /⟦skill:([^\]⟧\r\n]+)⟧/g,
    (_m, n) => {
      const name = String(n).replace(/["']/g, '')
      return `<span class="skill-chip" data-skill="${name}">◆ ${name}</span>`
    },
  )
  return withSkills.replace(
    /@(?:参考图|图)(\d+)/g,
    (_m, n) => `<span class="mention" data-mention="${n}">@图${n}</span>`,
  )
}

/** 从 contenteditable 的 innerHTML 提取纯文本。
 *
 *  IMPORTANT: this is the canonical text model — the concatenation of text
 *  nodes ONLY. Chromium's Range.toString() (used for live caret offsets) does
 *  NOT include `<br>` elements NOR div boundaries, and the SHOW_TEXT
 *  TreeWalker (used to map offsets back onto DOM positions) agrees with it.
 *  The previous version translated `<br>` into `\n` — desyncing the three
 *  views by one char per `<br>` and causing off-by-N errors: after the first
 *  mention (which inserts a `<br>`), every later `@图X` was anchored at the
 *  wrong offset and the chip ended up appended at the END of the whole prompt
 *  on a new line, no matter where the user typed `@`. */
export function htmlToText(html: string): string {
  if (!html) return ''
  let s = html.replace(/<br\s*\/?>/gi, '')
  // Skill 胶囊是整段 token 的可视化：span 被删一半/属性丢失时，
  // 里面的 `◆ 名字` 裸文本不能残留，统一还原成 token 再做完整性收尾。
  s = s.replace(/<span[^>]*data-skill="([^"]*)"[^>]*>.*?<\/span>/gi, (_m, n) => `⟦skill:${n}⟧`)
  s = s.replace(/<span[^>]*class="[^"]*skill-chip[^"]*"[^>]*>.*?<\/span>/gi, '')
  s = s.replace(/<[^>]+>/g, '')
  s = s.replace(/&nbsp;/g, ' ').replace(/&amp;/g, '&').replace(/&lt;/g, '<').replace(/&gt;/g, '>')
  // 残缺 token 收尾：胶囊只删了一半时会剩下 `⟦skill:名字` / `名字⟧` 碎片，
  // 统一清除，保证文本模型里要么是完整 token、要么什么都没有。
  s = s.replace(/⟦skill:[^\]⟧\r\n]*$/g, '')
  s = s.replace(/^[^\[⟦\r\n]*⟧/g, '')
  s = s.replace(/⟦skill:[^\]⟧\r\n]*[\r\n]/g, '')
  return s
}

/** Map an absolute character index (canonical text model: text nodes only) to
 *  the DOM text node + offset containing it. Falls back to the last text
 *  node's end for indices at/after the total length. */
function findTextPos(el: HTMLElement, abs: number): { node: Text; offset: number } | null {
  const walker = document.createTreeWalker(el, NodeFilter.SHOW_TEXT)
  let acc = 0
  let node: Node | null
  while ((node = walker.nextNode())) {
    const tn = node as Text
    const len = tn.data.length
    if (acc + len >= abs) return { node: tn, offset: abs - acc }
    acc += len
  }
  // abs may equal total length — fall back to last text node, end offset.
  const all: Text[] = []
  const w2 = document.createTreeWalker(el, NodeFilter.SHOW_TEXT)
  let n: Node | null
  while ((n = w2.nextNode())) all.push(n as Text)
  const last = all[all.length - 1]
  if (last) return { node: last, offset: last.data.length }
  return null
}

/** The text node that follows `node` in document order, or null. */
function nextTextNodeAfter(el: HTMLElement, node: Node): Text | null {
  const walker = document.createTreeWalker(el, NodeFilter.SHOW_TEXT)
  let n: Node | null
  let seen = false
  while ((n = walker.nextNode())) {
    if (seen) return n as Text
    if (n === node) seen = true
  }
  return null
}

/** Live caret offset in the canonical text model, or -1 when unavailable. */
function caretTextOffset(el: HTMLElement): number {
  const sel = window.getSelection()
  if (!sel || sel.rangeCount === 0) return -1
  try {
    const range = sel.getRangeAt(0)
    if (!el.contains(range.startContainer)) return -1
    const pre = document.createRange()
    pre.selectNodeContents(el)
    pre.setEnd(range.startContainer, range.startOffset)
    return pre.toString().length
  } catch {
    return -1
  }
}

/** True when the '@' at character index `idx` belongs to an EXISTING mention
 *  chip (an old @图N), not to a fresh query the user is typing. */
function isMentionChipAt(el: HTMLElement, text: string, idx: number): boolean {
  if (text[idx] !== '@') return false
  let node = findTextPos(el, idx)
  if (!node) return false
  // idx may sit on a node boundary (end of a text node); the '@' then starts
  // the NEXT text node (e.g. a chip that begins a new line).
  if (node.offset >= node.node.data.length) {
    const next = nextTextNodeAfter(el, node.node)
    if (!next) return false
    node = { node: next, offset: 0 }
  }
  if (node.node.data[node.offset] !== '@') return false
  const parent = node.node.parentElement
  return !!parent && !!parent.closest('.mention')
}

/** Find the active "@query" segment the caret sits inside.
 *
 *  The query is anchored AT THE CARET: scan backward from the caret across
 *  non-whitespace, non-'@' characters until we hit a fresh '@' (that's the
 *  anchor). The scan stops — no menu — at whitespace, at an old mention chip,
 *  or at the start of the text. Anchoring at the caret (instead of at the
 *  LAST '@' in the text) is what keeps the popup correct when older @图N
 *  chips exist later in the prompt: previously the query was anchored at the
 *  last chip, the wrong range got deleted, and the new chip was appended at
 *  the end of the whole prompt on a new line.
 *
 *  `fallbackCaret` is used only when the live selection is unavailable (React
 *  re-render / IME); handleInput keeps it fresh from the last known caret. */
function findActiveQuery(el: HTMLElement, text: string, fallbackCaret: number): { atStart: number; atEnd: number; query: string } {
  const live = caretTextOffset(el)
  const caretPos = live >= 0 ? Math.min(live, text.length) : Math.min(fallbackCaret, text.length)
  if (caretPos <= 0) return { atStart: -1, atEnd: -1, query: '' }

  for (let i = caretPos; i > 0; i--) {
    const ch = text[i - 1]
    if (ch === '@') {
      if (isMentionChipAt(el, text, i - 1)) return { atStart: -1, atEnd: -1, query: '' }
      return { atStart: i - 1, atEnd: caretPos, query: text.slice(i, caretPos) }
    }
    if (/\s/.test(ch ?? '')) return { atStart: -1, atEnd: -1, query: '' }
  }
  return { atStart: -1, atEnd: -1, query: '' }
}

function plainToInnerHTML(plain: string): string {
  const lines = plain.split('\n')
  return lines.map((line, i) => {
    const html = i === lines.length - 1 && line === ''
      ? '<br>'
      : (line === '' ? '<br>' : mentionizeMarkup(line))
    return `<div>${html}</div>`
  }).join('') || '<div><br></div>'
}

type Action =
  | { type: 'set-prompt'; value: string }
  | { type: 'set-param'; key: keyof GenerateParams; value: GenerateParams[keyof GenerateParams] }
  | { type: 'apply-template'; id: TemplateId }
  | { type: 'set-refs'; refs: RefImage[] }
  | { type: 'set-status'; status: Status; error?: ComposerState['error']; requestId?: string }
  | { type: 'set-results'; results: GeneratedImage[]; entry?: HistoryEntry }
  | { type: 'set-history'; history: HistoryEntry[] }
  | { type: 'set-settings'; settings: AppSettings }
  | { type: 'set-preview'; entry: HistoryEntry | undefined }

const INITIAL_PARAMS: GenerateParams = { aspect: '1:1', resolution: 'auto', quality: 'auto', count: 1 }

const initialState: ComposerState = {
  prompt: '',
  params: INITIAL_PARAMS,
  refImages: [],
  status: 'idle',
  error: undefined,
  results: [],
  history: [],
  settings: DEFAULT_SETTINGS,
}

function reducer(state: ComposerState, action: Action): ComposerState {
  switch (action.type) {
    case 'set-prompt': return { ...state, prompt: action.value }
    case 'set-param': return { ...state, params: { ...state.params, [action.key]: action.value } }
    case 'apply-template': {
      const t = TEMPLATES.find((x) => x.id === action.id)
      return t ? { ...state, prompt: t.text } : state
    }
    case 'set-refs': return { ...state, refImages: action.refs }
    case 'set-status':
      return {
        ...state,
        status: action.status,
        error: action.error,
        activeRequestId: action.status === 'loading' ? action.requestId : undefined,
      }
    case 'set-results':
      return { ...state, results: action.results, previewEntry: action.entry, activeRequestId: undefined }
    case 'set-history': return { ...state, history: action.history }
    case 'set-settings': return { ...state, settings: action.settings }
    case 'set-preview': return { ...state, previewEntry: action.entry, results: action.entry?.images ?? [] }
    default: return state
  }
}

import type { RefImage } from '@shared/types'

const TOKEN_RE = /@(?:参考图|图)\s*(\d+)?/g

export interface ResolveResult {
  text: string
  referenced: number[]
  errors: string[]
  warnings: string[]
}

/**
 * Resolve @图N tokens in the user prompt.
 *
 * - Always send every uploaded image, in index order (positional alignment invariant).
 * - 1 image: strip tokens (no disambiguation needed; the model gets one image).
 * - 2+ images: replace `@图N` with `第N张参考图` so multi-image referents stay
 *   disambiguated in natural prose — and the literal sigil doesn't get baked
 *   into the generated image.
 */
export function resolvePrompt(raw: string, refs: RefImage[]): ResolveResult {
  const errors: string[] = []
  const warnings: string[] = []
  const referenced = new Set<number>()

  if (refs.length === 0) {
    // No images uploaded; just strip any stray tokens and report errors.
    const cleaned = raw.replace(TOKEN_RE, '').replace(/\s{2,}/g, ' ').trim()
    if (/@(参考图|图)/.test(raw)) {
      errors.push('提示词包含 @图N 但未上传参考图')
    }
    return { text: cleaned, referenced: [], errors, warnings }
  }

  if (refs.length === 1) {
    // Strip tokens; collapse whitespace.
    let cleaned = raw.replace(TOKEN_RE, (m, n) => {
      const idx = n ? parseInt(n, 10) : 1
      if (idx !== 1) {
        errors.push(`提示词引用了 @图${idx}，但只上传了 1 张参考图`)
      } else {
        referenced.add(1)
      }
      return ''
    })
    cleaned = cleaned.replace(/[ \t]{2,}/g, ' ').replace(/([一-龥]) ([一-龥])/g, '$1$2').trim()
    return { text: cleaned, referenced: Array.from(referenced), errors, warnings }
  }

  // 2+ images: substitute with `第N张参考图`.
  let text = raw.replace(TOKEN_RE, (_m, n) => {
    const idx = n ? parseInt(n, 10) : 1
    if (idx < 1 || idx > refs.length) {
      errors.push(`提示词引用了 @图${idx}，但只上传了 ${refs.length} 张参考图`)
      return ''
    }
    referenced.add(idx)
    return `第${idx}张参考图`
  })

  // Tidy stray spaces.
  text = text.replace(/[ \t]{2,}/g, ' ').trim()

  // Warn about un-referenced images.
  for (let i = 1; i <= refs.length; i++) {
    if (!referenced.has(i)) warnings.push(`图${i} 未在提示词中引用`)
  }

  return { text, referenced: Array.from(referenced), errors, warnings }
}

export interface SkillResolveResult {
  text: string
  referenced: string[]
  errors: string[]

export function resolveSkillPrompt(raw: string, skills: Skill[]): SkillResolveResult {
  const errors: string[] = []
  const referenced: string[] = []
  const seen = new Set<string>()
  const byName = new Map(skills.map((s) => [s.name, s]))
  // SKILL_TOKEN_RE 是全局正则（/g），用 .test 会受 lastIndex 污染，这里手写匹配。
  const TOKEN_RE = /⟦skill:([^\]⟧\r\n]+)⟧/g
  const text = raw.replace(TOKEN_RE, (m, name: string) => {
    const skill = byName.get(name)
    if (!skill) {
      errors.push(`引用的 Skill「${name}」已不存在，请删除红色胶囊后重试`)
      return ''
    }
    if (seen.has(name)) return ''
    seen.add(name)
    referenced.push(name)
    return `\n\n【Skill：${skill.name}】\n${skill.content}\n`
  })
  return { text: text.replace(/[ \t]{2,}/g, ' ').trim(), referenced, errors }
}

// 校验：编辑器里的哪些 skill token 已经失效（右上角删掉后输入框还留着胶囊）。
export function missingSkillNames(raw: string, skills: Skill[]): string[] {
  const names = new Set(skills.map((s) => s.name))
  const missing = new Set<string>()
  const TOKEN_RE = /⟦skill:([^\]⟧\r\n]+)⟧/g
  let m: RegExpExecArray | null
  while ((m = TOKEN_RE.exec(raw)) !== null) {
    const name = m[1]
    if (name && !names.has(name)) missing.add(name)
  }
  return Array.from(missing)
}

// 文本里实际引用的 skill 名（去重保序），历史 params 行 `· N skill` 计数用。
export function skillNamesIn(raw: string): string[] {
  const out: string[] = []
  const seen = new Set<string>()
  const TOKEN_RE = /⟦skill:([^\]⟧\r\n]+)⟧/g
  let m: RegExpExecArray | null
  while ((m = TOKEN_RE.exec(raw)) !== null) {
    const name = m[1]
    if (name && !seen.has(name)) {
      seen.add(name)
      out.push(name)
    }
  }
  return out
}

function usePrintSize(colRef: React.RefObject<HTMLDivElement | null>): number {
  const [size, setSize] = useState(400)
  useLayoutEffect(() => {
    const el = colRef.current
    if (!el) return
    const measure = () => {
      const row = el.parentElement
      const head = el.parentElement?.querySelector(':scope > .eyebrow')
      const rowH = row?.clientHeight ?? 400
      const headH = head instanceof HTMLElement ? head.offsetHeight : 28
      const w = el.clientWidth
      // 行高 - header - 列内 gap ≈ 正方形可用高度；列宽即宽度上限
      const next = Math.max(200, Math.min(560, w, rowH - headH - 2))
      setSize((prev) => (Math.abs(prev - next) < 1 ? prev : next))
    }
    measure()
    const ro = new ResizeObserver(measure)
    if (el.parentElement) ro.observe(el.parentElement)
    ro.observe(el)
    window.addEventListener('resize', measure)
    return () => {
      ro.disconnect()
      window.removeEventListener('resize', measure)
    }
  }, [colRef])
  return size
}

        idx={idx}
        setIdx={setIdx}

        onSaveCurrent={() => {
          if (!current) return
          void window.api.downloadHistoryFile(current.file)
        }}
      />
    )
  }

  return box
}

import type { Skill } from './types'

// Skill 胶囊 token：`⟦skill:NAME⟧`。选用生僻括号对做定界，
// 用户正常输入几乎不可能敲出来，回解析不会误伤正文。
// NAME 内不允许 `⟧` / `]` / 换行，清洗规则 main + renderer 共用同一份。
export const SKILL_TOKEN_RE = /⟦skill:([^\]⟧\r\n]+)⟧/g

export function normalizeSkillName(input: unknown): string {
  const s = typeof input === 'string' ? input : ''
  // `"`/`'` 去掉：名字会写进 HTML 属性 data-skill="NAME"，留引号会打破标签。
  // `&`/`<>` 去掉：渲染前要过 HTML escape，留着会导致胶囊匹配不上（回解析错位）。
  return s.replace(/[\r\n⟦⟧\]"'&<>]/g, '').trim().slice(0, 60)
}

// 单个 content 上限 200KB（约 10 万汉字），超限整条丢弃，
// 防止 settings.json 体积爆炸拖慢每次启动加载。
export const SKILL_CONTENT_MAX = 200_000

export function newSkillId(): string {
  try {
    // 浏览器/Electron 渲染进程：优先用 Web Crypto
    const c = (globalThis as unknown as { crypto?: { randomUUID?: () => string } }).crypto
    if (c?.randomUUID) return c.randomUUID()
  } catch { /* fall through */ }
  return Date.now().toString(36) + Math.random().toString(36).slice(2, 10)
}

// 与 normalizeModelList 同风格的清洗：去空/超长/超大 → 按名去重 → 兜底 []。
export function normalizeSkillList(input: unknown): Skill[] {
  if (!Array.isArray(input)) return []
  const seen = new Set<string>()
  const out: Skill[] = []
  for (const x of input) {
    if (!x || typeof x !== 'object') continue
    const r = x as Partial<Skill>
    const name = normalizeSkillName(r.name)
    const content = typeof r.content === 'string' ? r.content : ''
    if (!name || !content.trim() || content.length > SKILL_CONTENT_MAX) continue
    if (seen.has(name)) continue
    seen.add(name)
    out.push({
      id: typeof r.id === 'string' && r.id ? r.id : newSkillId(),
      name,
      content,
      sourceName: typeof r.sourceName === 'string' ? r.sourceName : '',
      charCount: typeof r.charCount === 'number' ? r.charCount : content.length,
      createdAt: typeof r.createdAt === 'number' ? r.createdAt : Date.now(),
    })
  }
  return out
}

// 从文件名推默认 skill 名：去扩展名 → 清洗 → 为空则回退。
export function skillNameFromFile(fileName: string): string {
  const base = fileName.replace(/\.[^.]+$/, '').replace(/[_-]+/g, ' ').trim()
  return normalizeSkillName(base) || '未命名Skill'
}
